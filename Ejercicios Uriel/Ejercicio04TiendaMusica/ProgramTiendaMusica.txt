using EjerciciosNet02.Ejercicio04TiendaMusica;

EjercicioTiendaMusica();
static void EjercicioTiendaMusica()
{
    TiendaMusica tienda = new TiendaMusica();
    bool activo=true;
    while (activo)
    {
        int input;
        Console.WriteLine("Lo que desee realizar:");
        Console.WriteLine("1:Ingresar un instrumento");
        Console.WriteLine("2:Eliminar un instrumento");
        Console.WriteLine("3:Verificar instrumentos por categoría");
        Console.WriteLine("4:Verificar precio total de instrumentos en stock");
        Console.WriteLine("Cualquier otra tecla para salir");
        input=Convert.ToInt16(Console.ReadLine()!);

        switch (input)
        {
            case 1:
                Console.WriteLine("Ingrese el nombre del instrumento:");
                string nombre = Console.ReadLine()!;
                Console.WriteLine("Ingrese la categoría del instrumento:");
                string categoria = Console.ReadLine()!;
                Console.WriteLine("Ingrese el precio del instrumento:");
                int precio = Convert.ToInt16(Console.ReadLine()!);
                Instrumentos instrumento = new Instrumentos(){Nombre=nombre,Categoria=categoria,Precio=precio};
                tienda.AgregarInstrumento(instrumento);
                break;
            case 2:
                Console.WriteLine("Ingrese el nombre del instrumento que desea eliminar:");
                string nombreEliminar = Console.ReadLine()!;
                var instrumentoEliminar = tienda.ListaInstrumentos.FirstOrDefault(i => i.Nombre == nombreEliminar);
                if (instrumentoEliminar != null)
                {
                    tienda.EliminarInstrumento(instrumentoEliminar);
                    Console.WriteLine("Instrumento eliminado correctamente.");
                }
                else
                {
                    Console.WriteLine("Instrumento no encontrado.");
                }
                break;
            case 3:
                Console.WriteLine("Ingrese la categoría de los instrumentos que desea verificar:");
                string categoriaVerificar = Console.ReadLine()!;
                tienda.IntrumentoPorCategoria(categoriaVerificar);
                break;
            case 4:
                tienda.PrecioTotalIntrumentos();
                break;
            default:
                activo=false;
                break;
        }
    }
}
